  <script>
const books = {
  Fiction: [
    {name:"Harry Potter", stock:2},
    {name:"The Alchemist", stock:1}
  ],

  Science: [
    {name:"Physics Fundamentals", stock:2},
    {name:"Biology Basics", stock:1}
  ],

  Technology: [
    {name:"Java Programming", stock:3},
    {name:"Web Development", stock:2}
  ],

  History: [
    {name:"World War II", stock:1},
    {name:"Indian History", stock:2}
  ]
};

const category = document.getElementById("category");
const book = document.getElementById("book");
const form = document.getElementById("reservationForm");
const table = document.getElementById("reservationTable");
const summary = document.getElementById("summary");
const searchInput = document.getElementById("searchInput");
const progressBar = document.getElementById("progressBar");

let reservations = JSON.parse(localStorage.getItem("reservations")) || [];

displayReservations();

// LOAD BOOKS
category.addEventListener("change", () => {

  book.innerHTML = `<option value="">Select Book</option>`;

  books[category.value].forEach(b => {

    let option = document.createElement("option");

    option.value = b.name;

    option.textContent = `${b.name} (${b.stock} left)`;

    if(b.stock <= 0){
      option.disabled = true;
    }

    book.appendChild(option);
  });
});

// PROGRESS BAR
form.addEventListener("input", () => {

  let filled = form.querySelectorAll("input,select");

  let count = 0;

  filled.forEach(f => {
    if(f.value !== ""){
      count++;
    }
  });

  let percent = (count / filled.length) * 100;

  progressBar.style.width = percent + "%";
});

// FORM SUBMIT
form.addEventListener("submit", (e) => {

  e.preventDefault();

  const name = document.getElementById("name").value;
  const usn = document.getElementById("usn").value;
  const dept = document.getElementById("dept").value;
  const email = document.getElementById("email").value;
  const contact = document.getElementById("contact").value;
  const date = document.getElementById("date").value;

  // VALIDATIONS
  let usnPattern = /^[A-Za-z0-9]+$/;
  let phonePattern = /^[0-9]{10}$/;

  if(!usnPattern.test(usn)){
    alert("Invalid USN");
    return;
  }

  if(!phonePattern.test(contact)){
    alert("Invalid Contact Number");
    return;
  }

  let today = new Date().toISOString().split("T")[0];

  if(date < today){
    alert("Date cannot be in past");
    return;
  }

  const reservation = {

    id: "LIB" + Date.now(),
    name,
    usn,
    dept,
    email,
    contact,
    category: category.value,
    book: book.value,
    date,
    duration: document.getElementById("duration").value,
    status: "Reserved"
  };

  reservations.push(reservation);

  localStorage.setItem("reservations", JSON.stringify(reservations));

  alert("Reservation Successful");

  displayReservations();

  showSummary(reservation);

  calculateMostBorrowed();

  downloadReceipt(reservation);

  form.reset();

  progressBar.style.width = "0%";
});

// DISPLAY TABLE
function displayReservations(){

  table.innerHTML = "";

  reservations.forEach((r,index) => {

    let row = `
      <tr>
        <td>${r.id}</td>
        <td>${r.name}</td>
        <td>${r.book}</td>
        <td>${r.category}</td>
        <td>${r.date}</td>
        <td class="reserved">${r.status}</td>

        <td>
          <button onclick="editReservation(${index})">Edit</button>

          <button onclick="cancelReservation(${index})">
            Cancel
          </button>
        </td>
      </tr>
    `;

    table.innerHTML += row;
  });
}

// SUMMARY
function showSummary(r){

  summary.innerHTML = `
    <p><b>Reservation ID:</b> ${r.id}</p>
    <p><b>Student:</b> ${r.name}</p>
    <p><b>Book:</b> ${r.book}</p>
    <p><b>Status:</b> ${r.status}</p>
  `;
}

// EDIT
function editReservation(index){

  let r = reservations[index];

  document.getElementById("name").value = r.name;
  document.getElementById("usn").value = r.usn;
  document.getElementById("dept").value = r.dept;
  document.getElementById("email").value = r.email;
  document.getElementById("contact").value = r.contact;
  document.getElementById("date").value = r.date;
  document.getElementById("duration").value = r.duration;

  reservations.splice(index,1);

  localStorage.setItem("reservations", JSON.stringify(reservations));

  displayReservations();
}

// CANCEL
function cancelReservation(index){

  if(confirm("Cancel Reservation?")){

    reservations[index].status = "Cancelled";

    localStorage.setItem("reservations", JSON.stringify(reservations));

    displayReservations();
  }
}

// SEARCH
searchInput.addEventListener("keyup", () => {

  let value = searchInput.value.toLowerCase();

  let rows = table.querySelectorAll("tr");

  rows.forEach(row => {

    let text = row.innerText.toLowerCase();

    row.style.display = text.includes(value) ? "" : "none";
  });
});

// THEME TOGGLE
document.getElementById("themeBtn").addEventListener("click", () => {

  document.body.classList.toggle("dark-mode");
});

// MOST BORROWED
function calculateMostBorrowed(){

  let count = {};

  reservations.forEach(r => {

    count[r.category] = (count[r.category] || 0) + 1;
  });

  let max = "";

  let maxCount = 0;

  for(let c in count){

    if(count[c] > maxCount){

      maxCount = count[c];

      max = c;
    }
  }

  document.getElementById("mostBorrowed").innerHTML =
    "📖 Most Borrowed Category: " + max;
}

// DOWNLOAD RECEIPT
function downloadReceipt(r){

  let content = `
Reservation Receipt

ID: ${r.id}
Student: ${r.name}
Book: ${r.book}
Category: ${r.category}
Date: ${r.date}
Status: ${r.status}
  `;

  let blob = new Blob([content], {type:"text/plain"});

  let a = document.createElement("a");

  a.href = URL.createObjectURL(blob);

  a.download = "receipt.txt";

  a.click();
}

calculateMostBorrowed();
