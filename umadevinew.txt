<script>

/* PAGE SWITCH */
function showPage(id){
document.querySelectorAll(".container").forEach(p=>p.style.display="none");
document.getElementById(id).style.display="block";
if(id==="appointmentsPage") displayAppointments();
}

/* DATA */
const doctors={
"General Medicine":["Dr. Rajiv Sharma","Dr. Suman Mehta"],
"Pediatrics":["Dr. Priya Reddy","Dr. Samba Shivarao"],
"Orthopedics":["Dr. Ravi Kumar","Dr. Gabbar Singh"],
"Dermatology":["Dr. Aman Patel","Dr. Nisha Pati"]
};

const allSlots=["9:00 AM","10:00 AM","12:00 PM","3:00 PM","4:00 PM"];

let appointments=
JSON.parse(localStorage.getItem("medibook_appointments"))||[];

/* ELEMENTS */
const form=document.getElementById("appointmentForm");
const dept=document.getElementById("department");
const doctor=document.getElementById("doctor");
const slots=document.getElementById("slots");
const dateInput=document.getElementById("appointmentDate");
const search=document.getElementById("search");
const table=document.getElementById("appointmentTable");

/* MIN DATE */
dateInput.min=new Date().toISOString().split("T")[0];

/* LOAD DOCTORS */
dept.onchange=()=>{
doctor.innerHTML="<option>Select Doctor</option>";
(doctors[dept.value]||[]).forEach(d=>{
doctor.innerHTML+=`<option>${d}</option>`;
});
};

/* LOAD SLOTS */
doctor.onchange=dateInput.onchange=()=>{
slots.innerHTML="<option>Select Slot</option>";

allSlots.forEach(s=>{
let booked=appointments.some(a=>
a.doctor===doctor.value &&
a.date===dateInput.value &&
a.slot===s &&
a.status==="Booked");

slots.innerHTML+=
`<option ${booked?"disabled":""}>
${s}${booked?" (Booked)":""}
</option>`;
});
};

/* VALIDATION */
function validate(){
let ok=true;

if(!patientName.value){nameErr.style.display="block";ok=false;}
else nameErr.style.display="none";

if(patientAge.value<1||patientAge.value>120){ageErr.style.display="block";ok=false;}
else ageErr.style.display="none";

if(!/^\S+@\S+\.\S+$/.test(patientEmail.value)){emailErr.style.display="block";ok=false;}
else emailErr.style.display="none";

if(!/^\d{10}$/.test(patientPhone.value)){phoneErr.style.display="block";ok=false;}
else phoneErr.style.display="none";

return ok;
}

/* BOOK */
form.onsubmit=e=>{
e.preventDefault();
if(!validate())return;

let app={
bookingID:"BK"+Date.now(),
name:patientName.value,
age:patientAge.value,
email:patientEmail.value,
phone:patientPhone.value,
department:dept.value,
doctor:doctor.value,
date:dateInput.value,
slot:slots.value,
status:"Booked"
};

appointments.push(app);
localStorage.setItem("medibook_appointments",JSON.stringify(appointments));

summaryBox.style.display="block";
summaryContent.innerHTML=`
<b>ID:</b> ${app.bookingID}<br>
<b>Name:</b> ${app.name}<br>
<b>Doctor:</b> ${app.doctor}<br>
<b>Date:</b> ${app.date}<br>
<b>Slot:</b> ${app.slot}
`;

form.reset();
};

/* DISPLAY TABLE */
function displayAppointments(data=appointments){

table.innerHTML=
`<tr>
<th>ID</th><th>Name</th><th>Doctor</th>
<th>Date</th><th>Slot</th>
<th>Status</th><th>Action</th>
</tr>`;

if(!data.length){
table.innerHTML+=`<tr><td colspan=7>No data</td></tr>`;
return;
}

data.forEach(app=>{
let i=appointments.indexOf(app);
table.innerHTML+=`
<tr>
<td>${app.bookingID}</td>
<td>${app.name}</td>
<td>${app.doctor}</td>
<td>${app.date}</td>
<td>${app.slot}</td>
<td>${app.status}</td>
<td>
<button class="cancel-btn"
onclick="cancelAppointment(${i})">
Cancel
</button>
</td>
</tr>`;
});
}

/* CANCEL */
function cancelAppointment(i){
if(!confirm("Cancel appointment?"))return;
appointments.splice(i,1);
localStorage.setItem("medibook_appointments",JSON.stringify(appointments));
displayAppointments();
}

/* SEARCH */
search.onkeyup=()=>{
let val=search.value.toLowerCase();
displayAppointments(
appointments.filter(a=>a.name.toLowerCase().includes(val))
);
};

/* THEME TOGGLE */
function toggleTheme(){
document.documentElement.classList.toggle("dark");

if(document.documentElement.classList.contains("dark")){
themeBtn.textContent="☀️ Light Mode";
localStorage.setItem("medibook_theme","dark");
}else{
themeBtn.textContent="🌙 Dark Mode";
localStorage.setItem("medibook_theme","light");
}
}

/* LOAD SAVED THEME */
if(localStorage.getItem("medibook_theme")==="dark"){
document.documentElement.classList.add("dark");
themeBtn.textContent="☀️ Light Mode";
}

/* INIT */
displayAppointments();
