<script>

const doctors = {

"General Medicine":[
"Dr. Priya",
"Dr. Rajesh"
],

"Pediatrics":[
"Dr. Meena",
"Dr. Suresh"
],

"Orthopedics":[
"Dr. Arvind",
"Dr. Kavitha"
],

"Dermatology":[
"Dr. Deepa",
"Dr. Farhan"
]

};

const allSlots = [

"09:00 AM",
"09:30 AM",
"10:00 AM",
"10:30 AM",
"11:00 AM",
"11:30 AM",
"02:00 PM",
"02:30 PM"

];

let appointments =
JSON.parse(localStorage.getItem("appointments")) || [];

let selectedSlot = "";

let editId = null;

function get(id){
    return document.getElementById(id);
}

function showMessage(text,color){

    let msg = get("message");

    msg.style.display = "block";
    msg.style.background = color;
    msg.style.color = "white";

    msg.innerHTML = text;

    setTimeout(()=>{
        msg.style.display = "none";
    },3000);
}

function showSection(section){

    get("bookSection").style.display = "none";
    get("historySection").style.display = "none";

    if(section=="book"){

        get("bookSection").style.display = "block";
    }

    else{

        get("historySection").style.display = "block";

        renderAppointments();
    }
}

function toggleTheme(){

    document.body.classList.toggle("dark");
}

function loadDoctors(){

    let dept = get("department").value;

    let doctor = get("doctor");

    doctor.innerHTML =
    '<option value="">Select</option>';

    if(doctors[dept]){

        doctors[dept].forEach(d=>{

            doctor.innerHTML +=
            `<option>${d}</option>`;

        });
    }
}

function loadSlots(){

    let doctor = get("doctor").value;

    let date = get("date").value;

    let slotsDiv = get("slots");

    slotsDiv.innerHTML = "";

    if(!doctor || !date) return;

    let bookedSlots =
    appointments
    .filter(a=>

        a.doctor==doctor &&
        a.date==date &&
        a.status=="Booked"

    )
    .map(a=>a.slot);

    allSlots.forEach(slot=>{

        let btn =
        document.createElement("button");

        btn.innerHTML = slot;

        btn.className =
        bookedSlots.includes(slot)
        ? "slot booked"
        : "slot available";

        if(bookedSlots.includes(slot)){

            btn.disabled = true;
        }

        btn.onclick = ()=>selectSlot(slot);

        slotsDiv.appendChild(btn);

    });
}

function selectSlot(slot){

    selectedSlot = slot;

    document
    .querySelectorAll(".slot")
    .forEach(btn=>

        btn.classList.remove("selected")

    );

    [...document.querySelectorAll(".slot")]
    .find(btn=>btn.innerHTML==slot)
    .classList.add("selected");

    showSummary();

    showMessage(
    "Slot Available",
    "green"
    );
}

function showSummary(){

    get("summary").innerHTML = `

    <p><b>Name:</b>
    ${get("name").value}</p>

    <p><b>Doctor:</b>
    ${get("doctor").value}</p>

    <p><b>Date:</b>
    ${get("date").value}</p>

    <p><b>Slot:</b>
    ${selectedSlot}</p>

    `;
}

function validate(){

    if(get("name").value==""){

        showMessage(
        "Enter Patient Name",
        "red"
        );

        return false;
    }

    if(get("age").value<1 ||
       get("age").value>120){

        showMessage(
        "Invalid Age",
        "red"
        );

        return false;
    }

    if(!/^\d{10}$/.test(get("contact").value)){

        showMessage(
        "Invalid Contact Number",
        "red"
        );

        return false;
    }

    if(get("email").value==""){

        showMessage(
        "Enter Email",
        "red"
        );

        return false;
    }

    if(get("date").value <
       new Date().toISOString().split("T")[0]){

        showMessage(
        "Date Cannot Be In Past",
        "red"
        );

        return false;
    }

    if(selectedSlot==""){

        showMessage(
        "Select Slot",
        "red"
        );

        return false;
    }

    return true;
}

function bookAppointment(){

    if(!validate()) return;

    let appointment = {

        id: editId ||
        "MB"+Date.now(),

        token:
        Math.floor(Math.random()*100),

        name:get("name").value,

        age:get("age").value,

        gender:get("gender").value,

        contact:get("contact").value,

        email:get("email").value,

        department:get("department").value,

        doctor:get("doctor").value,

        date:get("date").value,

        slot:selectedSlot,

        reason:get("reason").value,

        status:"Booked"
    };

    if(editId){

        let index =
        appointments.findIndex(
        a=>a.id==editId
        );

        appointments[index] =
        appointment;

        editId = null;

        showMessage(
        "Appointment Updated Successfully",
        "green"
        );
    }

    else{

        appointments.push(
        appointment
        );

        showMessage(
        "Appointment Booked Successfully",
        "green"
        );
    }

    localStorage.setItem(
    "appointments",
    JSON.stringify(appointments)
    );

    renderAppointments();

    alert(

    `Booking Confirmed

Booking ID:
${appointment.id}

Token Number:
${appointment.token}`

    );

    resetForm();
}

function renderAppointments(){

    let search =
    get("search").value.toLowerCase();

    let data =
    appointments.filter(a=>

        a.name
        .toLowerCase()
        .includes(search)

    );

    let body =
    get("tableBody");

    if(data.length==0){

        body.innerHTML =

        `<tr>
        <td colspan="8">
        No Appointments
        </td>
        </tr>`;

        return;
    }

    body.innerHTML =
    data.map(a=>`

    <tr>

    <td>${a.id}</td>

    <td>${a.name}</td>

    <td>${a.doctor}</td>

    <td>${a.department}</td>

    <td>${a.date}</td>

    <td>${a.slot}</td>

    <td>${a.status}</td>

    <td>

    ${

    a.status=="Booked"

    ?

    `

    <button
    class="btn primary"
    onclick="editAppointment('${a.id}')">

    Edit

    </button>

    <button
    class="btn danger"
    onclick="cancelAppointment('${a.id}')">

    Cancel

    </button>

    `

    :

    "Cancelled"

    }

    </td>

    </tr>

    `).join("");
}

function editAppointment(id){

    let a =
    appointments.find(x=>x.id==id);

    editId = id;

    showSection("book");

    get("name").value = a.name;
    get("age").value = a.age;
    get("gender").value = a.gender;
    get("contact").value = a.contact;
    get("email").value = a.email;
    get("department").value = a.department;

    loadDoctors();

    get("doctor").value = a.doctor;
    get("date").value = a.date;
    get("reason").value = a.reason;

    loadSlots();

    selectedSlot = a.slot;

    showSummary();
}

function cancelAppointment(id){

    let confirmCancel =
    confirm("Cancel Appointment?");

    if(!confirmCancel) return;

    let a =
    appointments.find(x=>x.id==id);

    a.status = "Cancelled";

    localStorage.setItem(
    "appointments",
    JSON.stringify(appointments)
    );

    renderAppointments();

    showMessage(
    "Appointment Cancelled Successfully",
    "orange"
    );
}

function resetForm(){

    [
    "name",
    "age",
    "contact",
    "email",
    "reason",
    "date"
    ].forEach(id=>

        get(id).value=""

    );

    [
    "gender",
    "department",
    "doctor"
    ].forEach(id=>

        get(id).value=""

    );

    get("slots").innerHTML = "";

    get("summary").innerHTML = "";

    selectedSlot = "";
}

renderAppointments();

</script>
