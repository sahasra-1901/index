/* -----------------------------
   DOCTOR DATA
----------------------------- */

const doctors = {

"General Medicine":[
"Dr. Priya",
"Dr. Rajesh"
],

"Pediatrics":[
"Dr. Meena",
"Dr. Suresh"
],

"Orthopedics":[
"Dr. Arvind",
"Dr. Kavitha"
],

"Dermatology":[
"Dr. Deepa",
"Dr. Farhan"
]

};

/* -----------------------------
   ALL AVAILABLE TIME SLOTS
----------------------------- */

const allSlots = [

"09:00 AM",
"09:30 AM",
"10:00 AM",
"10:30 AM",
"11:00 AM",
"11:30 AM",
"02:00 PM",
"02:30 PM"

];

/* -----------------------------
   LOAD APPOINTMENTS FROM
   LOCAL STORAGE
----------------------------- */

let appointments =
JSON.parse(localStorage.getItem("appointments"))
|| [];

/* -----------------------------
   VARIABLES
----------------------------- */

let selectedSlot = "";

let editId = null;

/* -----------------------------
   SHORTCUT FUNCTION
   RETURNS ELEMENT USING ID
----------------------------- */

function get(id){
    return document.getElementById(id);
}

/* -----------------------------
   SHOW SUCCESS / ERROR MESSAGE
----------------------------- */

function showMessage(text,color){

    let msg = get("message");

    msg.style.display = "block";
    msg.style.background = color;
    msg.style.color = "white";

    msg.innerHTML = text;

    // HIDE MESSAGE AFTER 3 SECONDS
    setTimeout(()=>{

        msg.style.display = "none";

    },3000);
}

/* -----------------------------
   SHOW BOOK OR HISTORY SECTION
----------------------------- */

function showSection(section){

    // HIDE BOTH SECTIONS FIRST
    get("bookSection").style.display =
    "none";

    get("historySection").style.display =
    "none";

    // SHOW SELECTED SECTION
    if(section=="book"){

        get("bookSection").style.display =
        "block";
    }

    else{

        get("historySection").style.display =
        "block";

        renderAppointments();
    }
}

/* -----------------------------
   DARK / LIGHT THEME
----------------------------- */

function toggleTheme(){

    document.body.classList.toggle("dark");
}

/* -----------------------------
   LOAD DOCTORS BASED ON
   SELECTED DEPARTMENT
----------------------------- */

function loadDoctors(){

    let dept = get("department").value;

    let doctor = get("doctor");

    // RESET DOCTOR DROPDOWN
    doctor.innerHTML =
    '<option value="">Select</option>';

    // ADD DOCTORS OF SELECTED DEPARTMENT
    if(doctors[dept]){

        doctors[dept].forEach(d=>{

            doctor.innerHTML +=
            `<option>${d}</option>`;

        });
    }
}

/* -----------------------------
   LOAD AVAILABLE SLOTS
----------------------------- */

function loadSlots(){

    let doctor = get("doctor").value;

    let date = get("date").value;

    let slotsDiv = get("slots");

    slotsDiv.innerHTML = "";

    // STOP IF DOCTOR OR DATE NOT SELECTED
    if(!doctor || !date) return;

    // FIND ALREADY BOOKED SLOTS
    let bookedSlots =
    appointments
    .filter(a=>

        a.doctor==doctor &&
        a.date==date &&
        a.status=="Booked"

    )
    .map(a=>a.slot);

    // CREATE BUTTON FOR EACH SLOT
    allSlots.forEach(slot=>{

        let btn =
        document.createElement("button");

        btn.innerHTML = slot;

        btn.className =

        bookedSlots.includes(slot)

        ?

        "slot booked"

        :

        "slot available";

        // DISABLE BUTTON IF SLOT IS BOOKED
        if(bookedSlots.includes(slot)){

            btn.disabled = true;
        }

        // SELECT SLOT WHEN CLICKED
        btn.onclick = ()=>selectSlot(slot);

        slotsDiv.appendChild(btn);
    });
}

/* -----------------------------
   SELECT SLOT
----------------------------- */

function selectSlot(slot){

    selectedSlot = slot;

    // REMOVE OLD SELECTED CLASS
    document
    .querySelectorAll(".slot")
    .forEach(btn=>

        btn.classList.remove("selected")

    );

    // ADD SELECTED CLASS TO CLICKED SLOT
    [...document.querySelectorAll(".slot")]
    .find(btn=>btn.innerHTML==slot)
    .classList.add("selected");

    // SHOW SUMMARY
    showSummary();

    showMessage(
    "Slot Available",
    "green"
    );
}

/* -----------------------------
   SHOW APPOINTMENT SUMMARY
----------------------------- */

function showSummary(){

    // CALCULATE ESTIMATED WAITING TIME
    let waitingTime =
    (allSlots.indexOf(selectedSlot)+1)
    * 15;

    get("summary").innerHTML = `

    <p><b>Name:</b>
    ${get("name").value}</p>

    <p><b>Doctor:</b>
    ${get("doctor").value}</p>

    <p><b>Date:</b>
    ${get("date").value}</p>

    <p><b>Slot:</b>
    ${selectedSlot}</p>

    <p><b>Estimated Waiting Time:</b>
    ${waitingTime} Minutes</p>

    `;
}

/* -----------------------------
   FORM VALIDATION
----------------------------- */

function validate(){

    // CHECK NAME
    if(get("name").value==""){

        showMessage(
        "Enter Patient Name",
        "red"
        );

        return false;
    }

    // CHECK AGE
    if(get("age").value<1 ||
       get("age").value>120){

        showMessage(
        "Invalid Age",
        "red"
        );

        return false;
    }

    // CHECK CONTACT NUMBER
    if(!/^\d{10}$/.test(
       get("contact").value)){

        showMessage(
        "Invalid Contact Number",
        "red"
        );

        return false;
    }

    // CHECK EMAIL
    if(get("email").value==""){

        showMessage(
        "Enter Email",
        "red"
        );

        return false;
    }

    // CHECK PAST DATE
    if(get("date").value <
       new Date().toISOString().split("T")[0]){

        showMessage(
        "Date Cannot Be In Past",
        "red"
        );

        return false;
    }

    // CHECK SLOT SELECTION
    if(selectedSlot==""){

        showMessage(
        "Select Slot",
        "red"
        );

        return false;
    }

    return true;
}

/* -----------------------------
   BOOK OR UPDATE APPOINTMENT
----------------------------- */

function bookAppointment(){

    // STOP IF VALIDATION FAILS
    if(!validate()) return;

    // CREATE APPOINTMENT OBJECT
    let appointment = {

        id:
        editId ||
        "MB"+Date.now(),

        token:
        Math.floor(Math.random()*100),

        name:get("name").value,

        age:get("age").value,

        gender:get("gender").value,

        contact:get("contact").value,

        email:get("email").value,

        department:get("department").value,

        doctor:get("doctor").value,

        date:get("date").value,

        slot:selectedSlot,

        reason:get("reason").value,

        status:"Booked"
    };

    // UPDATE EXISTING APPOINTMENT
    if(editId){

        let index =
        appointments.findIndex(
        a=>a.id==editId
        );

        appointments[index] =
        appointment;

        editId = null;

        showMessage(
        "Appointment Updated Successfully",
        "green"
        );
    }

    // ADD NEW APPOINTMENT
    else{

        appointments.push(appointment);

        showMessage(
        "Appointment Booked Successfully",
        "green"
        );
    }

    // SAVE TO LOCAL STORAGE
    localStorage.setItem(

    "appointments",

    JSON.stringify(appointments)

    );

    // REFRESH TABLE
    renderAppointments();

    // SHOW CONFIRMATION MODAL
    openModal(

    `Booking Confirmed <br><br>

    Booking ID:
    ${appointment.id}

    <br><br>

    Token Number:
    ${appointment.token}`

    );

    // RESET FORM
    resetForm();
}

/* -----------------------------
   DISPLAY APPOINTMENTS
----------------------------- */

function renderAppointments(){

    // GET SEARCH AND FILTER VALUES
    let search =
    get("search").value.toLowerCase();

    let deptFilter =
    get("filterDept").value;

    let doctorFilter =
    get("filterDoctor").value;

    let dateFilter =
    get("filterDate").value;

    let sortOption =
    get("sortOption").value;

    // FILTER APPOINTMENTS
    let data =
    appointments.filter(a=>{

    return (

        a.name.toLowerCase()
        .includes(search)

        &&

        (deptFilter==""
        ||

        a.department==deptFilter)

        &&

        (doctorFilter==""
        ||

        a.doctor==doctorFilter)

        &&

        (dateFilter==""
        ||

        a.date==dateFilter)

    );

});

/* SORT BY NAME */
if(sortOption=="name"){

    data.sort((a,b)=>

    a.name.localeCompare(b.name)

    );
}

/* SORT BY DATE */
if(sortOption=="date"){

    data.sort((a,b)=>

    new Date(a.date)
    -
    new Date(b.date)

    );
}

/* COUNT APPOINTMENTS BY DEPARTMENT */
let counts = {};

appointments.forEach(a=>{

    if(a.status=="Booked"){

        counts[a.department] =

        (counts[a.department] || 0)+1;
    }
});

/* DISPLAY DEPARTMENT COUNTS */
get("departmentCount").innerHTML =

Object.entries(counts)
.map(c=>

`<p><b>${c[0]}</b> :
${c[1]} Appointments</p>`

).join("");

/* FIND NEXT UPCOMING APPOINTMENT */
let upcoming =

appointments

.filter(a=>

a.status=="Booked"

&&

new Date(a.date)>=new Date()

)

.sort((a,b)=>

new Date(a.date)
-
new Date(b.date)

)[0];

/* SHOW NEXT APPOINTMENT */
if(upcoming){

    get("nextAppointment").innerHTML =

    `<h3>

    Next Appointment:

    ${upcoming.name}

    -

    ${upcoming.date}

    -

    ${upcoming.slot}

    </h3>`;
}

else{

    get("nextAppointment").innerHTML =

    "<h3>No Upcoming Appointments</h3>";
}

let body =
get("tableBody");

/* SHOW MESSAGE IF NO APPOINTMENTS */
if(data.length==0){

    body.innerHTML =

    `<tr>

    <td colspan="8">

    No Appointments

    </td>

    </tr>`;

    return;
}

/* DISPLAY APPOINTMENTS IN TABLE */
body.innerHTML =

data.map(a=>`

<tr>

<td>${a.id}</td>

<td>${a.name}</td>

<td>${a.doctor}</td>

<td>${a.department}</td>

<td>${a.date}</td>

<td>${a.slot}</td>

<td>${a.status}</td>

<td>

${

a.status=="Booked"

?

`

<button
class="btn primary"
onclick="editAppointment('${a.id}')">

Edit

</button>

<button
class="btn danger"
onclick="cancelAppointment('${a.id}')">

Cancel

</button>

`

:

"Cancelled"

}

</td>

</tr>

`).join("");
}

/* -----------------------------
   EDIT APPOINTMENT
----------------------------- */

function editAppointment(id){

    // FIND APPOINTMENT
    let a =
    appointments.find(x=>x.id==id);

    editId = id;

    // OPEN BOOK SECTION
    showSection("book");

    // FILL FORM WITH OLD DATA
    get("name").value = a.name;
    get("age").value = a.age;
    get("gender").value = a.gender;
    get("contact").value = a.contact;
    get("email").value = a.email;
    get("department").value = a.department;

    loadDoctors();

    get("doctor").value = a.doctor;

    get("date").value = a.date;

    get("reason").value = a.reason;

    loadSlots();

    selectedSlot = a.slot;

    showSummary();
}

/* -----------------------------
   CANCEL APPOINTMENT
----------------------------- */

function cancelAppointment(id){

    let confirmCancel =
    confirm("Cancel Appointment?");

    if(!confirmCancel) return;

    // FIND APPOINTMENT
    let a =
    appointments.find(x=>x.id==id);

    // CHANGE STATUS
    a.status = "Cancelled";

    // SAVE UPDATED DATA
    localStorage.setItem(

    "appointments",

    JSON.stringify(appointments)

    );

    renderAppointments();

    showMessage(

    "Appointment Cancelled Successfully",

    "orange"

    );
}

   RESET FORM
----------------------------- */

    [

    "name",
    "age",
    "contact",
    "email",
    "reason",
    "date"

    ].forEach(id=>

        get(id).value=""

    );

    // RESET DROPDOWNS
    [

    "gender",
    "department",
    "doctor"

    ].forEach(id=>

        get(id).value=""

    );

    // CLEAR SLOT AREA
    get("slots").innerHTML = "";

    // CLEAR SUMMARY
    get("summary").innerHTML = "";

    selectedSlot = "";

    updateProgress();
}

/* -----------------------------
   UPDATE FORM PROGRESS BAR
----------------------------- */

function updateProgress(){

    let total = 8;

    let filled = 0;

    // CHECK FILLED FIELDS
    [

    "name",
    "age",
    "gender",
    "contact",
    "email",
    "department",
    "doctor",
    "date"

    ].forEach(id=>{

        if(get(id).value!=""){

            filled++;
        }
    });

    // CALCULATE PERCENTAGE
    let percent =

    Math.floor((filled/total)*100);

    // UPDATE PROGRESS BAR
    get("progressBar").style.width =
    percent+"%";

    get("progressBar").innerHTML =
    percent+"%";
}

function createWindow() {
  const win = new BrowserWindow({
    width: 1200,
    height: 800,
    webPreferences: {
      nodeIntegration: true,
      contextIsolation: false
    }
  });

  win.loadFile('index.html'); // 👈 YOUR HTML FILE
}
